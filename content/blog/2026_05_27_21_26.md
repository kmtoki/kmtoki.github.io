- neovimのGUIのneovideをためしている。昔はIMEに問題があったので秒でやめたが、どうやらその問題はプラグインで回避できるらしい。ならええか。goneovimは開発終了したっぽく最新のneovimではエラーがでるので乗り換える方向で。長年くそお世話になりました。


```vimscript
Plug 'sevenc-nanashi/neov-ime.nvim' "for Neovide's IME

if exists("g:neovide")
  set guifont=Ricty_Diminished:h17
  let g:neovide_refresh_rate = 120
  let g:neovide_position_animation_length = 0.15
  let g:neovide_scroll_animation_length = 0
  let g:neovide_cursor_animation_length = 0

  augroup ime_input
      autocmd!
      autocmd InsertLeave * execute "let g:neovide_input_ime=v:false"
      autocmd InsertEnter * execute "let g:neovide_input_ime=v:true"
      autocmd CmdlineLeave [/\?] execute "let g:neovide_input_ime=v:false"
      autocmd CmdlineEnter [/\?] execute "let g:neovide_input_ime=v:true"
  augroup END
endif
```
